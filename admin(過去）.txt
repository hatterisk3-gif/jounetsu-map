      window.calcRidges = (coords, dir, widthCm) => {
        if(!dir || !widthCm || widthCm <= 0 || !coords || coords.length < 3) return "--";
        const gCoords = coords.map(pt => new google.maps.LatLng(pt.lat, pt.lng));
        let areaM2 = google.maps.geometry.spherical.computeArea(gCoords);
        let edges = [];
        for (let i = 0; i < gCoords.length; i++) {
          let p1 = gCoords[i]; let p2 = gCoords[(i + 1) % gCoords.length];
          let length = google.maps.geometry.spherical.computeDistanceBetween(p1, p2);
          let heading = google.maps.geometry.spherical.computeHeading(p1, p2);
          let absH = Math.abs(heading); if (absH > 90) absH = 180 - absH; 
          edges.push({ length, heading: absH });
        }
        let targetEdges = [];
        if (dir === '南北') targetEdges = edges.filter(e => e.heading <= 45);
        else if (dir === '東西') targetEdges = edges.filter(e => e.heading > 45);
        if (targetEdges.length === 0) targetEdges = edges;
        targetEdges.sort((a, b) => b.length - a.length);
        let baseLength = targetEdges[0].length;
        let fieldWidthM = areaM2 / baseLength;
        let ridgeWidthM = widthCm / 100;
        return Math.floor((fieldWidthM * 0.95) / ridgeWidthM);
      };

      function setupMapSearch() {
        const input = document.getElementById('mapSearchInput');
        const searchBox = new google.maps.places.SearchBox(input);

        map.addListener('bounds_changed', () => {
          searchBox.setBounds(map.getBounds());
        });

        searchBox.addListener('places_changed', () => {
          const places = searchBox.getPlaces();
          if (places.length == 0) return;

          const bounds = new google.maps.LatLngBounds();
          places.forEach(place => {
            if (!place.geometry || !place.geometry.location) return;
            if (place.geometry.viewport) {
              bounds.union(place.geometry.viewport);
            } else {
              bounds.extend(place.geometry.location);
            }
          });
          map.fitBounds(bounds);
        });
      }

      window.execAttr = (id) => {
        const p=loadedPolygons[id];
        if(p.isMarker){
           const n=document.getElementById('rnIn').value; 
           const f=document.getElementById('rnFunc').value;
           if(!n)return; 
           p.name=n; p.signFunction=f; p.labelConfig.text=n; 
           p.marker.setMap(null); p.marker=createSignboardMarker(n,p.marker.getPosition(),p.color,id); 
           callGAS('updatePolygon',{id,name:n,signFunction:f,userName:currentUser});
        }
        else{
          const n=document.getElementById('edN').value, l=document.getElementById('edL').value, c=document.getElementById('edC').value, s=document.getElementById('edS').value;
          const t = p.toukiId; 
          const rDir=document.getElementById('edRDir').value, rW=document.getElementById('edRW').value;
          if(!n)return; p.name=n; p.location=l; p.condition=c; p.status=s; p.toukiId=t; p.ridgeDir=rDir; p.ridgeWidth=rW;
          const isU=(s==='未使用（返却）'||s==='未使用'), col=isU?'#999':p.color; p.polygon.setOptions({fillColor:col,strokeColor:col,fillOpacity:isU?0.5:0.3}); p.marker.setMap(null); p.marker=createLabelMarker(n,p.polygon.getPath().getArray(),p.color,p.area); 
          callGAS('updatePolygon',{id,name:n,location:l,condition:c,status:s,toukiId:t, ridgeDir:rDir, ridgeWidth:rW, userName:currentUser});

        if(bId===tId) return; if(!await customConfirm("マスタと履歴を統合しますか？")) {isMergeMode=false; return;}
        const bP=loadedPolygons[bId], tP=loadedPolygons[tId];
        if(tP.toukiId) bP.toukiId = bP.toukiId ? [...new Set((bP.toukiId+","+tP.toukiId).split(","))].join(",") : tP.toukiId;
